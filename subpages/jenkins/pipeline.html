the pipeline:

pipeline {
    agent {
        node {
          label "built-in"
		  //user custom workspace
          customWorkspace "userPath/Project"
        }
    }
	// variables:
    environment {
		// shorts below:
        //Progra~1 -it's folder "Program Files"
        //Progra~2 -it's folder "Program Files x(86)""
        
		_tool = "c:\\user\\path\\toolcli.exe"
        _CONFIG = "builtin://CWE Top 25 2023"
    }
	// blocks of 4 stages in the Jenkins 
    stages {
        stage('Check version') {
            steps { 
				// run command
                bat "\"${_tool}\" -version"
            }
        }
        stage('Clean report dir') {
            steps {
                dir ("path/report") {
                    deleteDir()
                }
            }
        }
        stage('Static analysis') {
            steps { 
				// run commands
                bat "echo ${_CONFIG}";
                bat "\"${_tool}\" -solution Sample.sln -config \"${_CONFIG}\" -report report"
            }
        }
        stage('ParasoftFinding Warnings') {
            steps {
                archiveArtifacts artifacts: 'report/*.html, report/*.xml, report/*.png, report/*.jpeg', fingerprint: false
                recordIssues tool: parasoftFindings(pattern: "report/report.xml")
            }
        }
    }
}
